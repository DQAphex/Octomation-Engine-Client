def sync_processes_to_db(diccionario_funciones):
  db = SessionLocal()

  try:
    for clave, data in diccionario_funciones.items():
      meta = data["metadata"]
      process_id = meta.get("id", clave)

      db_process = (
        db.query(BackgroundProcessModel)
        .filter(BackgroundProcessModel.id == process_id)
        .first()
      )

      if not db_process:
        db_process = BackgroundProcessModel(
          id=process_id,
          clave_modulo=clave,
          name=meta.get("name", process_id),
          description=meta.get("description", ""),
          category=meta.get("category", "general"),
          version=meta.get("version", "1.0.0"),
          enabled=meta.get("enabled", False)
        )
        db.add(db_process)
      else:
        db_process.clave_modulo = clave
        db_process.name = meta.get("name", db_process.name)
        db_process.description = meta.get(
          "description",
          db_process.description
        )
        db_process.category = meta.get(
          "category",
          db_process.category
        )
        db_process.version = meta.get(
          "version",
          db_process.version
        )

      db.flush()

      meta["id"] = db_process.id
      meta["enabled"] = db_process.enabled

      # ------------------------------------------------------------
      # Horarios
      # ------------------------------------------------------------

      for sched in meta.get("schedules", []):
        if not isinstance(sched, dict):
          continue

        hour = sched.get("hour")
        minute = sched.get("minute")

        if hour is None or minute is None:
          continue

        if not isinstance(hour, int) or not isinstance(minute, int):
          continue

        if not 0 <= hour <= 23 or not 0 <= minute <= 59:
          continue

        existing_schedule = (
          db.query(ProcessScheduleModel)
          .filter(
            ProcessScheduleModel.process_id == process_id,
            ProcessScheduleModel.hour == hour,
            ProcessScheduleModel.minute == minute
          )
          .first()
        )

        if not existing_schedule:
          db.add(
            ProcessScheduleModel(
              process_id=process_id,
              hour=hour,
              minute=minute
            )
          )

      # ------------------------------------------------------------
      # Opciones
      # ------------------------------------------------------------

      sync_options_to_db(
        db,
        process_id,
        meta.get("options", [])
      )

    db.commit()

  except Exception as e:
    db.rollback()
    print(
      f"Error al sincronizar procesos "
      f"en la base de datos: {e}"
    )

  finally:
    db.close()

def get_background_proceses():
  diccionario_funciones = {}

  for archivo in bp_dir.rglob("worker.py"):
    if archivo.name.startswith("__"):
      continue

    ruta_relativa = archivo.relative_to(bp_dir)
    archivo_metadata = archivo.parent / "metadata.json"
    metadata = {}

    if archivo_metadata.exists():
      try:
        with open(
          archivo_metadata,
          "r",
          encoding="utf-8"
        ) as f:
          metadata = json.load(f)
      except Exception as e:
        print(
          f"Error leyendo "
          f"{archivo_metadata}: {e}"
        )
        continue
    else:
      print(
        f"Advertencia: {archivo} "
        "no tiene metadata.json."
      )

    clave_modulo = str(
      ruta_relativa.with_suffix("")
    ).replace("\\", "/")

    partes_ruta = [
      bp_dir.name
    ] + list(
      ruta_relativa
      .with_suffix("")
      .parts
    )

    nombre_modulo_completo = ".".join(partes_ruta)

    spec = importlib.util.spec_from_file_location(
      nombre_modulo_completo,
      archivo
    )

    if not spec or not spec.loader:
      print(
        f"Advertencia: no se pudo crear "
        f"spec para {archivo}."
      )
      continue

    modulo = importlib.util.module_from_spec(spec)

    modulo.__package__ = (
      nombre_modulo_completo.rsplit(".", 1)[0]
    )

    sys.modules[nombre_modulo_completo] = modulo

    try:
      spec.loader.exec_module(modulo)

      func_run = getattr(modulo, "run", None)

      if callable(func_run):
        diccionario_funciones[clave_modulo] = {
          "run": func_run,
          "metadata": metadata
        }
      else:
        print(
          f"Advertencia: {archivo} "
          "no tiene una función 'run'."
        )

    except Exception as e:
      print(
        f"Error al cargar "
        f"{archivo}: {e}"
      )
      sys.modules.pop(
        nombre_modulo_completo,
        None
      )

  sync_processes_to_db(diccionario_funciones)

  return diccionario_funciones

def add_schedule_to_process(
  process_id: str,
  hour: int,
  minute: int
) -> dict:

  if not 0 <= hour <= 23:
    return {
      "error": (
        "La hora debe estar en el rango "
        "de 0 a 23."
      )
    }

  if not 0 <= minute <= 59:
    return {
      "error": (
        "El minuto debe estar en el rango "
        "de 0 a 59."
      )
    }

  db = SessionLocal()

  try:
    process = (
      db.query(BackgroundProcessModel)
      .filter(
        BackgroundProcessModel.id == process_id
      )
      .first()
    )

    if not process:
      return {
        "error": (
          f"Proceso con ID '{process_id}' "
          "no fue encontrado."
        )
      }

    existing_schedule = (
      db.query(ProcessScheduleModel)
      .filter(
        ProcessScheduleModel.process_id == process_id,
        ProcessScheduleModel.hour == hour,
        ProcessScheduleModel.minute == minute
      )
      .first()
    )

    if existing_schedule:
      return {
        "error": (
          f"El horario {hour:02d}:{minute:02d} "
          "ya existe para este proceso."
        )
      }

    new_schedule = ProcessScheduleModel(
      process_id=process_id,
      hour=hour,
      minute=minute
    )

    db.add(new_schedule)
    db.commit()
    db.refresh(new_schedule)

    return {
      "id": new_schedule.id,
      "process_id": new_schedule.process_id,
      "hour": new_schedule.hour,
      "minute": new_schedule.minute,
      "status": new_schedule.status.value,
      "message": "Horario agregado exitosamente."
    }

  except Exception as e:
    db.rollback()
    return {
      "error": (
        f"Error al guardar el horario: {str(e)}"
      )
    }

  finally:
    db.close()

def update_bp_option_values(
  option_id: str,
  values: list
) -> dict:

  db = SessionLocal()

  try:
    option = (
      db.query(BPOptionsModel)
      .filter(
        BPOptionsModel.id == option_id
      )
      .first()
    )

    if not option:
      return {
        "error": (
          f"Opción con ID '{option_id}' "
          "no fue encontrada."
        )
      }

    if not isinstance(values, list):
      return {
        "error": (
          "Los valores deben enviarse "
          "como una lista."
        )
      }

    if not option.multiple and len(values) > 1:
      return {
        "error": (
          "Esta opción solo permite "
          "un valor."
        )
      }

    for value in values:
      if value is None:
        return {
          "error": (
            "Los valores no pueden ser null."
          )
        }

      if not isinstance(value, str):
        value = str(value)

      if len(value) > 500:
        return {
          "error": (
            "Un valor supera los 500 caracteres."
          )
        }

    option.values.clear()

    for value in values:
      option.values.append(
        BPOptionValueModel(
          value=value
        )
      )

    db.commit()
    db.refresh(option)

    return {
      "id": option.id,
      "key": option.key,
      "multiple": option.multiple,
      "values": [
        {
          "id": item.id,
          "value": item.value
        }
        for item in option.values
      ],
      "message": (
        "Valores actualizados correctamente."
      )
    }

  except Exception as e:
    db.rollback()
    return {
      "error": (
        f"Error al actualizar la opción: "
        f"{str(e)}"
      )
    }
  finally:
    db.close()

import importlib
import inspect
from pathlib import Path
from typing import List
from fastapi import APIRouter, FastAPI
from fastapi.staticfiles import StaticFiles
from fastapi.templating import Jinja2Templates

class PluginManager:
  def __init__(self, app: FastAPI, project_root: Path):
    self.app = app
    self.project_root = project_root
    self.plugins_dir = self.project_root / "plugins"
    self.global_static = self.project_root / "static"
    self.global_templates = self.project_root / "templates"
    
    self.template_dirs: List[Path] = []
    self.loaded_plugins: List[str] = []
    self.loaded_plugins_dirs: List[str] = []

  def get_loaded_routers(self):
    return self.loaded_plugins_dirs
  
  def discover_and_load(self):
    """
    Punto de entrada principal. Inicia el escaneo recursivo y establece 
    los montajes globales como fallback (evitando shadowing).
    """
    if not self.plugins_dir.exists() or not self.plugins_dir.is_dir():
      raise FileNotFoundError(f"Directorio de plugins no encontrado: {self.plugins_dir}")

    # 1. Escaneo topológico de dominios y submódulos (recursivo)
    self._scan_recursively(self.plugins_dir)

    # 2. Montaje estático global (Resolución de fallback en ASGI)
    if self.global_static.exists() and self.global_static.is_dir():
      self.app.mount(
        "/static",
        StaticFiles(directory=str(self.global_static)),
        name="static_global",
      )

  def _scan_recursively(self, current_path: Path):
    """
    Recorre el árbol de directorios ignorando artefactos de Python y sistemas,
    delegando la evaluación de recursos cuando detecta un paquete válido.
    """
    for path in current_path.iterdir():
      if path.is_dir() and not path.name.startswith((".", "_", "__")):
        # Prevenir escaneo en carpetas de datos internas del plugin
        if path.name in ["data", "background_processes", "backgroundProcesses", "fails"]:
          continue
          
        self._evaluate_plugin_node(path)
        self._scan_recursively(path)

  def _evaluate_plugin_node(self, plugin_path: Path):
    """
    Evalúa un nodo del directorio. Construye espacios de nombres dinámicos 
    basados en la ruta relativa desde la raíz de plugins.
    """
    # Ej: Posición actual: plugins/medioambiente/riles
    # rel_path = medioambiente/riles
    rel_path = plugin_path.relative_to(self.plugins_dir)
    namespace_parts = rel_path.parts
    
    # Si estamos en la raíz de plugins, no hay espacio de nombres
    if not namespace_parts:
      return

    # Construcción de identificadores dinámicos
    # url_namespace -> medioambiente/riles
    # name_namespace -> medioambiente_riles
    url_namespace = "/".join(namespace_parts)
    name_namespace = "_".join(namespace_parts)

    # --- 1. Montaje de Estáticos Aislados ---
    plugin_static = plugin_path / "static"
    if plugin_static.exists() and plugin_static.is_dir():
      mount_name = f"static_{name_namespace}"
      url_prefix = f"/static/plugins/{url_namespace}"
      
      self.app.mount(
        url_prefix,
        StaticFiles(directory=str(plugin_static)),
        name=mount_name,
      )

    # --- 2. Recolección de Plantillas ---
    for template_folder in ["pages", "templates"]:
      t_dir = plugin_path / template_folder
      if t_dir.exists() and t_dir.is_dir():
        self.template_dirs.append(t_dir)

    # --- 3. Carga Dinámica de Routers (Convención por Homonimia) ---
    # Busca un archivo .py que coincida con el nombre de la carpeta (ej. riles.py)
    # y opcionalmente chequea un main.py como fallback
    entry_points = [
      plugin_path / f"{plugin_path.name}.py",
      plugin_path / "main.py"
    ]

    for entry_point in entry_points:
      if entry_point.exists() and entry_point.is_file():
        # Transforma plugins/medioambiente/riles/riles.py -> plugins.medioambiente.riles.riles
        module_path = ".".join(["plugins"] + list(namespace_parts) + [entry_point.stem])
        self._import_routers_from_module(module_path, name_namespace)
        break # Si encuentra el router primario, detiene la búsqueda en esta carpeta

  def _import_routers_from_module(self, module_path: str, plugin_id: str):
    try:
      module = importlib.import_module(module_path)
      for attr_name, attr_value in inspect.getmembers(module):
        if isinstance(attr_value, APIRouter):
          if getattr(attr_value, "_registered_in_app", False):
            continue
          self.app.include_router(attr_value)
          setattr(attr_value, "_registered_in_app", True)
          self.loaded_plugins.append(f"[{plugin_id}] Montado: {module_path}:{attr_name}")
          self.loaded_plugins_dirs.append(f"{module_path}:{attr_name}")
    except Exception as e:
      print(f"❌ Error crítico en módulo {module_path}: {e}")


  def get_jinja_templates(self) -> Jinja2Templates:
    """
    Retorna la instancia de Jinja2 con resolución de fallback.
    Prioridad 1: Plantillas locales del plugin (/plugins/.../pages/)
    Prioridad 2: Plantillas globales (/templates/)
    """
    all_dirs = []
    
    # Inyección de directorios descubiertos en orden de precedencia
    all_dirs.extend([str(d) for d in self.template_dirs])

    if self.global_templates.exists() and self.global_templates.is_dir():
      all_dirs.append(str(self.global_templates))

    return Jinja2Templates(directory=all_dirs)

import asyncio
from apscheduler.schedulers.asyncio import AsyncIOScheduler
from .bp_crawler import (
  SessionLocal,
  BackgroundProcessModel,
  ProcessScheduleModel,
  get_background_proceses,
)

# Instancia global del scheduler con zona horaria definida
scheduler = AsyncIOScheduler(timezone="America/Santiago")


def make_sync_execution_wrapper(func):
  """Envuelve funciones normales (def) para que corran en un thread secundario sin bloquear asyncio."""

  async def wrapper():
    loop = asyncio.get_running_loop()
    await loop.run_in_executor(None, func)

  return wrapper


def register_single_job(process, sched) -> bool:
  """Registra o actualiza un job individual en el scheduler."""
  diccionario_funciones = get_background_proceses()

  if process.clave_modulo not in diccionario_funciones:
    return False

  raw_func = diccionario_funciones[process.clave_modulo].get("run")
  if not raw_func:
    return False

  func_to_run = (
      raw_func
      if asyncio.iscoroutinefunction(raw_func)
      else make_sync_execution_wrapper(raw_func)
  )
  job_id = f"job_{process.id}_{sched.id}"

  scheduler.add_job(
      func_to_run,
      "cron",
      hour=sched.hour,
      minute=sched.minute,
      id=job_id,
      misfire_grace_time=60,
      replace_existing=True,
  )
  print(
      f"Job actualizado/agregado: '{job_id}'"
      f" ({sched.hour:02d}:{sched.minute:02d})"
  )
  return True


def remove_single_job(process_id: int, schedule_id: int):
  """Remueve un job específico del scheduler si existe."""
  job_id = f"job_{process_id}_{schedule_id}"
  if scheduler.get_job(job_id):
    scheduler.remove_job(job_id)
    print(f"Job removido del scheduler: '{job_id}'")


def load_schedules_from_db():
  """Carga inicial de todos los horarios habilitados al arrancar la app."""
  db = SessionLocal()
  try:
    processes = (
        db.query(BackgroundProcessModel)
        .filter(BackgroundProcessModel.enabled == True)
        .all()
    )

    for process in processes:
      schedules = (
          db.query(ProcessScheduleModel)
          .filter(ProcessScheduleModel.process_id == process.id)
          .all()
      )

      for sched in schedules:
        register_single_job(process, sched)
  except Exception as e:
    print(f"Error cargando horarios iniciales: {e}")
  finally:
    db.close()
