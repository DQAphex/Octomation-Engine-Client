import fitz
import io
import re
import pdfplumber

class File_Runtime():

  def __init__(self,pdf_bytes: bytes):
    self.pdf_bytes=pdf_bytes
    self.tables = []
    self.folio = None
    self.folios_gd = None
    self.fecha_emision = None
    self.nombre_razon_social = None
    self.rut_razon_social = None
    self.excel_export = []
  def obtener_tablas(self):
    with pdfplumber.open(io.BytesIO(self.pdf_bytes)) as pdf:
      for pagina in pdf.pages:
        self.tables = pagina.extract_tables()
        return self.tables
        
  def buscar_tablas(self, palabra_clave):

    for tabla in self.tables:
      for fila in tabla:
        for celda in fila:
          if celda == palabra_clave:
            return tabla  

  def obtener_folios_gd(self) -> list[str]:
    tabla = self.buscar_tablas("FOLIO") #fila = ['TIPO GUÍA DESPACHO', 'FOLIO', 'FECHA DE EMISIÓN', 'FECHA DE RECEPCIÓN']
    folios = tabla[2][1]
    self.folios_gd = folios.split("\n")
    return folios.split("\n")
  
  def obtener_folio_certificado(self) -> list[str]:
    tabla = self.buscar_tablas("Nº FOLIO:") #fila = ['TIPO GUÍA DESPACHO', 'FOLIO', 'FECHA DE EMISIÓN', 'FECHA DE RECEPCIÓN']
    folio = tabla[0][2].split(" ")[1]
    self.folio = folio
    return folio
  
  def obtener_fecha_emision(self) -> list[str]:
    tabla = self.buscar_tablas("Nº FOLIO:") #fila 3 = ['SANTIAGO', '07-09-2026', None]]
    fecha_emision = tabla[2][1]
    self.fecha_emision = fecha_emision
    return fecha_emision
  
  def obtener_razon_social(self) -> list[str]:
    tabla = self.buscar_tablas("INFORMACIÓN DE INSTITUCIÓN QUE ENTREGA PRODUCTOS")
    fila_datos = tabla[1][1] #fila = ['NOMBRE O RAZÓN SOCIAL', 'RENDIC HERMANOS S.A.\n81537600-5\nCERRO EL PLOMO 5680, LAS CONDES\nJORGE SAEZ CORREA\n10034176-K']
    lista_datos_importantes = fila_datos.split("\n")
    nombre_razon_social = lista_datos_importantes[0]
    rut_razon_social = lista_datos_importantes[1]
    self.nombre_razon_social = nombre_razon_social
    self.rut_razon_social = rut_razon_social
    return [self.nombre_razon_social,self.rut_razon_social]

  def generar_tabla_excel(self):
    print(self.folios_gd)
    for folio_gd in self.folios_gd:
      self.excel_export.append([self.folio,folio_gd,self.rut_razon_social,self.nombre_razon_social,self.fecha_emision])
    return self.excel_export
  
  def orchestrator_extract_data(self):

    self.obtener_tablas()
    self.obtener_folio_certificado()
    self.obtener_fecha_emision()
    self.obtener_razon_social()
    self.obtener_folios_gd()
    self.generar_tabla_excel()

    return

def leer_pdf(pdf_bytes: bytes) -> str:
  with pymupdf.open(
    stream=pdf_bytes,
    filetype="pdf"
  ) as pdf:
    return "".join(
      pagina.get_text()
      for pagina in pdf
    )


def obtener_folio(texto: str) -> str | None:
  patron = (
    r"N[º°]\s*FOLIO\s*:\s*"
    r"(?:\n\s*)?"
    r"N?[º°]?\s*(\d+)"
  )

  resultado = re.search(
    patron,
    texto,
    re.IGNORECASE
  )

  if resultado:
    return resultado.group(1)

  return None


def generar_nombre_archivo(folio: str) -> str:
  folio_formateado = f"{int(folio):05d}"

  return f"FOLIO_N_{folio_formateado}.pdf"


def procesar_correos():
  carpetas_outlook, carpetas_destino = getFunctionalData()
  print(carpetas_outlook, carpetas_destino)

  pythoncom.CoInitialize()
  try:
    cantidad_pdf = 0
    outlook = win32com.client.Dispatch("Outlook.Application")
    namespace = outlook.GetNamespace("MAPI")
    inbox = namespace.GetDefaultFolder(6)

    fecha_desde = (
      datetime.now() - timedelta(days=2)
    ).replace(
      hour=0,
      minute=0,
      second=0,
      microsecond=0
    )

    for nombre_carpeta in carpetas_outlook:
      carpeta_destino = carpetas_destino[0]

      try:
        carpeta = inbox.Folders[nombre_carpeta]
      except Exception:
        print(
          f"❌ No se encontró la carpeta "
          f"'{nombre_carpeta}'"
        )
        continue

      print(
        f"✓ Carpeta encontrada: {nombre_carpeta}"
      )
      print(
        f"✓ Destino: {carpeta_destino}"
      )

      os.makedirs(
        carpeta_destino,
        exist_ok=True
      )

      items = carpeta.Items
      items.Sort("[ReceivedTime]", True)

      for message in items:
        try:
          recibido = message.ReceivedTime

          print(
            f"Correo: {message.Subject} | "
            f"ReceivedTime: {recibido}"
          )

          if recibido.tzinfo is not None:
            recibido = recibido.replace(
              tzinfo=None
            )

          if recibido < fecha_desde:
            print(
              "⏹ Correo anterior al límite. "
              "Deteniendo búsqueda."
            )
            break

          print(
            f"Procesando correo: {message.Subject}"
          )

          if message.Attachments.Count == 0:
            continue

          for attachment in message.Attachments:
            try:
              nombre_original = attachment.FileName

              if not nombre_original.lower().endswith(".pdf"):
                continue

              print(
                f"  PDF encontrado: "
                f"{nombre_original}"
              )

              with tempfile.NamedTemporaryFile(
                  suffix=".pdf",
                  delete=False
                ) as temp:
                  temp_path = temp.name

              try:
                attachment.SaveAsFile(temp_path)

                with open(temp_path, "rb") as archivo:
                  pdf_bytes = archivo.read()

                texto = leer_pdf(pdf_bytes)
                fr = File_Runtime(pdf_bytes=pdf_bytes)
                fr.orchestrator_extract_data()

                excel_guias = ExcelFile(file_path=carpetas_destino[0])
                excel_guias.export_to_excel(table=fr.excel_export)
                
                folio = fr.folio

              finally:
                if os.path.exists(temp_path):
                  os.remove(temp_path)

              if not folio:
                print(
                  "  ⚠ No se encontró el folio."
                )
                continue

              print(
                f"  ✓ Folio encontrado: {folio}"
              )

              nombre_archivo = (
                generar_nombre_archivo(folio)
              )

              ruta_destino = os.path.join(
                carpeta_destino,
                nombre_archivo
              )

              if os.path.exists(ruta_destino):
                print(
                  f"  ⚠ El archivo ya existe: "
                  f"{nombre_archivo}"
                )
                continue

              attachment.SaveAsFile(
                ruta_destino
              )

              print(
                f"  ✓ Guardado en: "
                f"{ruta_destino}"
              )

              cantidad_pdf += 1

            except Exception as e:
              print(
                f"  ❌ Error procesando adjunto "
                f"'{attachment.FileName}': {e}"
              )

          print()

        except Exception as e:
          print(
            f"❌ Error procesando correo "
            f"'{message.Subject}': {e}"
          )

    print(
      "========================================"
    )
    print(
      f"✓ Proceso terminado. "
      f"PDF descargados: {cantidad_pdf}"
    )
    print(
      "========================================"
    )

  finally:
    pythoncom.CoUninitialize()

import os
from openpyxl import load_workbook,Workbook

class ExcelFile:
  def __init__(
    self,
    file_path: str,
    file_name: str = "gestion_guias_despacho.xlsx"
  ):
    self.file_path = os.path.join(file_path, file_name)

    if os.path.exists(self.file_path):
      self.wb = load_workbook(self.file_path)
    else:
      self.wb = Workbook()

    self.ws = self.wb.active

  def export_to_excel(self, table):
    print(table)
    for row in table:
      self.ws.append(row)

    self.wb.save(self.file_path)
